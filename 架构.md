```text
Game
  world: WorldState
  cache: WorldCache

WorldState
  buildings[BuildingId]: Building
  building_map: BuildingMap
  workers[]: Worker
  ground_items[Cell]: Storage
  delivery_tasks[]: DeliveryTask
  belt_cargo[CargoId]: {delivery → DeliveryTask, from, to, progress}
  smart_objects: SmartObjects
  pollution_operations[Cell]: {radius, strength}
  building_ids_by_owner[OwnerId] → BuildingId[]
  building_ids_by_group[Group] → BuildingId[]
  built_ids_by_kind[BuildingKind] → BuildingId[]

Building
  id: BuildingId
  phase: BLUEPRINT | BUILT | RECOVERING | DEMOLISHING
  inventory: Storage
  state: {kind, owner, employee, wage, task, growth, sell_prices, ...}
  id → BuildingMap.placements[id]
  id → SmartObjects.instances["building:<id>"]

BuildingMap
  placements[BuildingId]: {origin: Cell, cells: Cell[]}
  cells[Cell] → BuildingId[]
  origins[Cell] → BuildingId[]
  chunks[Chunk] → BuildingId[]

ItemId = int | String
Goods = {item: ItemId, amount: float, pollution: float}

Storage
  items[ItemId]: float
  wastewater_load: float

DeliveryTask
  id, item: ItemId, amount, pollution
  source: Cell, target: Cell, receiver_id → BuildingId
  owner → WorkerId
  bound, picked, on_belt, payment, freight

SmartObjects
  definitions: SmartObjectDefinitions
    buildings[BuildingKind]: ObjectDefinition
    blueprints[BuildingKind]: ObjectDefinition
    demolitions[BuildingKind]: ObjectDefinition
    resources[ResourceKind]: ObjectDefinition
    delivery, deposit, stock: ObjectDefinition
  instances[ObjectId]: SmartObjectInstance
  endpoints[SlotId]: Endpoint
  slots[SlotId]: SlotState

ObjectDefinition
  kind, domain, materials[ItemId]: float
  products[]: ItemId
  interactions[]: SlotDefinition

SlotDefinition
  action, method, name, offsets[], shared, gate, items[]

SmartObjectInstance
  id: ObjectId
  definition → ObjectDefinition
  source → Building.state | Storage.items | Resource | DeliveryTask
  position: Cell
  slot_ids[] → SmartObjects.endpoints

Endpoint
  id: SlotId
  instance → SmartObjectInstance
  definition → SlotDefinition
  state → SmartObjects.slots[id]
  position: Cell
  items[], materials[]

SlotState
  owner → WorkerId
  source → SmartObjectInstance.source
  position, identity, claim, occupied, shared

Worker
  id: WorkerId
  name, pos, age_days, skill
  employer → BuildingId
  money, payroll, loan_principal, loan_interest
  items[ItemId], equipment[ItemId], water, gun_uses, backpack_uses
  trade_cargo[]: Goods
  delivery_tasks[] → WorldState.delivery_tasks
  hunger, fatigue, pollution, clothes_pollution
  mood: WorkerMood
  buffs: WorkerBuffs
  activity: WorkerStateTree

WorkerStateTree
  active: Activity?
  decision_wait, previous_limits, inertia_group, inertia_hours, drink_elapsed
  select() → ActivitySelection

Utility
  ActivitySelection
    intents[]: ActivityIntent
    evaluation: ActivityEvaluation
      context: WorkerContext
        game → Game
        worker → Worker
        paths[Cell]: Cell[]
    next_option() → ActivityOption

ActivityOption
  record: ActivityRecord
  approach_position, approach_path[], slots[] → SlotId[], score, data

Activity
  record → ActivityRecord
  source_identity → Building.state | DeliveryTask | Resource
  handles[]: {key → SlotId, identity, claim, owner → WorkerId}
  path[], destination, phase, elapsed, age
  Work | Life | Economy | Delivery

ActivityRecord
  kind, target: Cell, data
  method → data.method

WorldCache
  pollution: Pollution

Pollution
  cells[Cell]: float
  revision: int
```
